To find duplicates in your dataset using Python with pandas, you can check for exact duplicate rows across all columns or subset duplicates based on specific key columns (such as Station and SampleDate).

Here are the primary ways to check for, inspect, and remove duplicates in hab_surface_compiled.csv:

1. Check for Exact Duplicate Rows
To count and display full row duplicates:

Python
import pandas as pd

# Load dataset
df = pd.read_csv("hab_surface_compiled.csv")

# Count exact duplicate rows across all columns
num_duplicates = df.duplicated().sum()
print(f"Total exact duplicate rows: {num_duplicates}")

# Display the duplicate rows (including all occurrences)
duplicate_rows = df[df.duplicated(keep=False)]
print(duplicate_rows)
2. Check for Duplicates Based on Specific Key Columns
In environmental time-series datasets, duplicates often occur when a station has multiple entries recorded for the exact same sampling date or depth layer.

Python
# Key identifier columns for a sample
key_cols = ["Station", "SampleDate"]

# Count duplicates based on key columns
key_duplicates_count = df.duplicated(subset=key_cols).sum()
print(
    f"Duplicates based on {key_cols}: {key_duplicates_count}"
)

# View all duplicate entries sorted together for comparison
key_duplicates = df[df.duplicated(subset=key_cols, keep=False)].sort_values(
    by=key_cols
)
print(key_duplicates)

Keep the first occurrence and drop the rest:

Python

some of the phytoplankton counts are split into multiple entries, how would i combine those

To combine multiple entries recorded for the same station and date into a single combined entry, you use Pandas' groupby() and agg() functions.

Depending on how you want to combine the different parameters:

Phytoplankton counts (H_triquetra_cells_L, P_minimum_cells_L) are summed together ('sum').

Environmental parameters (Temperature_C, Salinity_ppt) can be averaged ('mean') or taken from the shallowest depth ('first').

Metadata (Station, Region, SampleDate, etc.) is retained.

Python Code: Combining Multiple Entries Per Station & Date
Python
import pandas as pd

# Load dataset
df = pd.read_csv("hab_surface_compiled.csv")

# 1. Sort by shallowest depth first so metadata/depth originates from the surface layer
df = df.sort_values(
    by=["Station", "SampleDate", "TS_Depth_m"], ascending=True
)

# 2. Group by Station and SampleDate, specifying how to aggregate each column
df_combined = (
    df.groupby(["Station", "SampleDate"], as_index=False)
    .agg(
        {
            # Metadata columns (take the first/shallowest entry)
            "Region": "first",
            "Latitude": "first",
            "Longitude": "first",
            "Year": "first",
            "Month": "first",
            "Season": "first",
            "TS_Depth_m": "min",  # Keeps shallowest depth
            "Phyto_Layer": "first",
            # Environmental parameters (average across split entries)
            "Temperature_C": "mean",
            "Salinity_ppt": "mean",
            # Phytoplankton cell counts (sum split counts together)
            "H_triquetra_cells_L": "sum",
            "P_minimum_cells_L": "sum",
        }
    )
)

# Save the combined dataset to a new CSV
df_combined.to_csv("hab_surface_combined_counts.csv", index=False)

To merge and clean the datasets so that you only keep dates/samples that have both phytoplankton and T/S data available, you filter out missing values (NaN / null) for temperature or salinity and aggregate any split cell counts for each station and date.

Here is the complete Python code to accomplish this step-by-step:

Python Code: Clean and Merge (Inner Join)
Python
import pandas as pd

# 1. Load dataset
df = pd.read_csv("hab_surface_compiled.csv")

# 2. Keep only rows where BOTH Temperature AND Salinity are present (non-null)
# (Phytoplankton counts are complete, so requiring non-null T/S filters to complete matches)
df_valid = df[df["Temperature_C"].notna() & df["Salinity_ppt"].notna()].copy()

# 3. Sort by station, date, and depth to ensure surface-most metadata is prioritized
df_sorted = df_valid.sort_values(
    by=["Station", "SampleDate", "TS_Depth_m"], ascending=True
)

# 4. Group by Station and SampleDate to aggregate counts and surface T/S measurements
df_merged = df_sorted.groupby(["Station", "SampleDate"], as_index=False).agg(
    {
        # Geographic and temporal metadata (taking surface layer value)
        "Region": "first",
        "Latitude": "first",
        "Longitude": "first",
        "Year": "first",
        "Month": "first",
        "Season": "first",
        "TS_Depth_m": "min",  # Surface-most sampling depth
        "Phyto_Layer": "first",
        # T/S Environmental Parameters
        "Temperature_C": "mean",
        "Salinity_ppt": "mean",
        # Phytoplankton cell counts (combining split entries)
        "H_triquetra_cells_L": "sum",
        "P_minimum_cells_L": "sum",
    }
)

# 5. Save clean merged dataset
df_merged.to_csv("hab_merged_clean.csv", index=False)

print(
    f"Successfully merged. Total records with complete T/S and Phytoplankton: {len(df_merged)}"
)
Alternative: If You Have Two Separate CSV Files
If your phytoplankton data and T/S data are currently in two separate CSV files (e.g., phyto_data.csv and ts_data.csv), you perform an inner join on Station and SampleDate:

Python
import pandas as pd

# Load separate files
df_phyto = pd.read_csv("phyto_data.csv")
df_ts = pd.read_csv("ts_data.csv")

# 1. Deduplicate T/S data to keep closest to surface
df_ts_surface = df_ts.sort_values(by=["Station", "SampleDate", "TS_Depth_m"])\
                      .drop_duplicates(subset=["Station", "SampleDate"], keep="first")

# 2. Combine split phytoplankton counts per station/date
df_phyto_combined = df_phyto.groupby(["Station", "SampleDate"], as_index=False).agg({
    "H_triquetra_cells_L": "sum",
    "P_minimum_cells_L": "sum"
})

# 3. Inner join (keeps ONLY dates where both datasets overlap)
df_merged = pd.merge(
    df_ts_surface,
    df_phyto_combined,
    on=["Station", "SampleDate"],
    how="inner"
)

df_merged.to_csv("hab_merged_clean.csv", index=False)

Python Code: Comparing Sampling Frequency Across Datasets
Python
import matplotlib.pyplot as plt
import pandas as pd

# Load dataset (or load separate df_phyto and df_ts files if you have them)
df = pd.read_csv("hab_surface_compiled.csv")

# 1. Filter and calculate unique sampling dates per year for T/S data
df_ts = df[df["Temperature_C"].notna() & df["Salinity_ppt"].notna()]
ts_counts = (
    df_ts.groupby(["Year", "Station"])["SampleDate"]
    .nunique()
    .unstack(fill_value=0)
)

# 2. Filter and calculate unique sampling dates per year for Phytoplankton data
df_phyto = df[
    df["H_triquetra_cells_L"].notna() & df["P_minimum_cells_L"].notna()
]
phyto_counts = (
    df_phyto.groupby(["Year", "Station"])["SampleDate"]
    .nunique()
    .unstack(fill_value=0)
)

fig, (ax1, ax2) = plt.subplots(

    "LE5.5": "#cc79a7",
    "LE5.5-W": "#b58d63",
    "WE4.2": "#f0e442",
}

# Plot Temperature & Salinity Dataset
for station in ts_counts.columns:
    ax1.plot(
        ts_counts.index,
        ts_counts[station],
        marker="o",
        markersize=4,
        label=station,
        linewidth=1.5,
        color=station_colors.get(station),

ax1.legend(title="Station", bbox_to_anchor=(1.01, 1), loc="upper left")

for station in phyto_counts.columns:

import matplotlib.pyplot as plt
import pandas as pd

import matplotlib.pyplot as plt
import pandas as pd

# 1. Load dataset
df = pd.read_csv("hab_surface_compiled.csv")

# 2. Categorize data availability per sampling event (Station + Date)
has_ts = df["Temperature_C"].notna() & df["Salinity_ppt"].notna()
has_phyto = df["H_triquetra_cells_L"].notna() & df["P_minimum_cells_L"].notna()


def categorize(row_idx):
    if has_ts[row_idx] and has_phyto[row_idx]:
        return "Both T/S & Phyto"
    elif has_ts[row_idx] and not has_phyto[row_idx]:
        return "Just T/S"
    elif not has_ts[row_idx] and has_phyto[row_idx]:
        return "Just Phyto"
    else:
        return "Neither"


df["Data_Category"] = [categorize(i) for i in df.index]

# 3. Pivot counts by Station and Data Category
station_counts = (
    df.groupby(["Station", "Data_Category"]).size().unstack(fill_value=0)
)

# Ensure all categories are present in order
categories_order = ["Both T/S & Phyto", "Just Phyto", "Just T/S"]

    if cat not in station_counts.columns:
        station_counts[cat] = 0

    "CB1.1",

    "CB5.2",

    "LE5.5-W",

import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns

# 1. Load data and select matched records
df = pd.read_csv("hab_surface_compiled.csv")

    df["H_triquetra_cells_L"].notna() & df["P_minimum_cells_L"].notna()
].copy()

# 2. Reshape into long format
df_long = pd.melt(
    df_matched,
    id_vars=["Station", "Region", "SampleDate"],
    value_vars=["H_triquetra_cells_L", "P_minimum_cells_L"],
    var_name="Species",
    value_name="Abundance_cells_L",
)

df_long["Species"] = df_long["Species"].map(
    {
        "H_triquetra_cells_L": "Heterocapsa triquetra",
        "P_minimum_cells_L": "Prorocentrum minimum",
    }
)

df_nonzero = df_long[df_long["Abundance_cells_L"] > 0].copy()

# Palette & Region Order

    "Upper Bay": "#1f77b4",

    "Lower Bay": "#2ca02c",
}

plt.figure(figsize=(12, 6))

ax = sns.boxplot(
    data=df_nonzero,
    x="Species",
    y="Abundance_cells_L",
    hue="Region",
    hue_order=region_order,
    palette=palette,
    showmeans=True,
    meanprops={
        "marker": "o",
        "markerfacecolor": "red",
        "markeredgecolor": "black",
        "markersize": "7",
    },
)

plt.tight_layout()
